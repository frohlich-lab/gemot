# Runner dependencies shared by every example's deepagents driver.
#
# The petab-mcp server itself must also be installed (AMICI + pyPESTO are core deps):
#     pip install -e .                         # from the repo root (needs swig + a C++ compiler)
#
# and an OpenRouter API key must be exported so the driver can reach the model:
#     export OPENROUTER_API_KEY=sk-or-...      # https://openrouter.ai/keys
#
# The --driver scripted mode (the no-LLM harness self-test) needs only the petab-mcp
# server (none of the packages below). Regenerating an example's synthetic data
# (generate_data.py) needs only amici + petab, which the server install already provides.

# NOTE: deepagents 0.6.12 requires the langchain 1.x line (langchain-core>=1.4.8), so
# langchain-openai must also be on 1.x — the old 0.3.0 pin caps langchain-core at <0.4 and
# makes the set unresolvable / import-broken (langgraph 1.x expects langchain-core 1.x).
deepagents==0.6.12
langchain-mcp-adapters==0.3.0
langchain-openai>=1.3,<2

# Arize Phoenix agent tracing (ON by default; disable at runtime with --no-phoenix).
# Traces every LLM + MCP tool call the agent makes as OpenTelemetry spans viewable in the
# Phoenix UI. arize-phoenix-otel + the LangChain instrumentor EXPORT the spans; the full
# `arize-phoenix` package provides the local in-process UI the runner launches when no
# PHOENIX_COLLECTOR_ENDPOINT is set.
# arize-phoenix >=18 provides the run_experiment / Datasets / Prompt-registry API that
# examples/run_experiments.py uses (client.experiments / client.datasets / client.prompts).
arize-phoenix>=18
arize-phoenix-otel>=0.6
openinference-instrumentation-langchain>=0.1

# Weights & Biases metrics logging for the model×temperature sweep (opt-in via run.py
# --wandb). Cluster-friendly alternative analysis UI to Phoenix Experiments: logs one
# small W&B run per (example × model × temperature × repeat) with the judge outcome; the
# free tier meters run storage (tiny here), and `WANDB_MODE=offline` + `wandb sync`
# survives an egress-restricted cluster. Credentials via `wandb login` (~/.netrc).
wandb>=0.18
