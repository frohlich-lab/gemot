# Runner dependencies shared by every example's deepagents driver.
#
# The petab-mcp server itself must also be installed (AMICI + pyPESTO are core deps):
#     pip install -e .                         # from the repo root (needs swig + a C++ compiler)
#
# and an OpenRouter API key must be exported so the driver can reach the model:
#     export OPENROUTER_API_KEY=sk-or-...      # https://openrouter.ai/keys
#
# The --driver scripted mode (the no-LLM harness self-test) needs only the petab-mcp
# server (none of the packages below). Regenerating an example's synthetic data
# (generate_data.py) needs only amici + petab, which the server install already provides.

langchain-mcp-adapters==0.3.0
langchain-openai==0.3.0

# `arize-phoenix` package provides the local in-process UI the runner launches when no
# PHOENIX_COLLECTOR_ENDPOINT is set.
